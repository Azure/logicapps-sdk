param(
    [Parameter(Mandatory)][string] $BaseUri,
    [Parameter(Mandatory)][string] $HostDirectory,
    [Parameter(Mandatory)][string] $DefinitionsDirectory,
    [Parameter(Mandatory)][string] $OutputPath
)
$ErrorActionPreference = 'Stop'
$uri = [Uri]$BaseUri
if (-not $uri.IsLoopback) { throw 'This probe is restricted to an isolated local host.' }
$status = Invoke-RestMethod "$BaseUri/admin/host/status"
if ($status.state -ne 'Running') { throw "Host is not running: $($status.state)" }
$listener = @(Get-NetTCPConnection -State Listen -LocalPort $uri.Port)
if (-not $listener.Count) { throw 'The host has no local listener.' }

$cases = @(
    @{ name = 'native-arithmetic'; workflow = 'NativeProbe'; input = '{}'; expected = '3' },
    @{ name = 'condition-true'; workflow = 'ConditionProbe'; input = '3'; expected = 'yes' },
    @{ name = 'condition-false'; workflow = 'ConditionProbe'; input = '4'; expected = 'no' },
    @{ name = 'template-condition-control-true'; workflow = 'ConditionTemplateControl'; input = '3'; expected = 'yes' },
    @{ name = 'template-condition-control-false'; workflow = 'ConditionTemplateControl'; input = '4'; expected = 'no' },
    @{ name = 'encoded-json'; workflow = 'EncodedJsonProbe'; input = '{}'; expected = [Convert]::ToBase64String([Text.Encoding]::UTF8.GetBytes('{"Next":4,"Label":"A"}')) },
    @{ name = 'json-intrinsic'; workflow = 'JsonIntrinsicProbe'; input = '{}'; expected = '{"LABEL":"A"}' }
)
$results = foreach ($case in $cases) {
    $callbackPath = "/runtime/webhooks/workflow/api/management/workflows/$($case.workflow)/triggers/manual/listCallbackUrl?api-version=2018-11-01"
    $callback = Invoke-WebRequest -Method Post ($BaseUri + $callbackPath) -SkipHttpErrorCheck
    if ($callback.StatusCode -eq 200) {
        $callbackUri = [Uri](($callback.Content | ConvertFrom-Json).value)
        if (-not $callbackUri.IsLoopback -or $callbackUri.Port -ne $uri.Port) {
            throw 'The callback is not served by the isolated local host.'
        }
        $response = Invoke-WebRequest -Method Post $callbackUri -ContentType 'application/json' `
            -Body $case.input -SkipHttpErrorCheck -TimeoutSec 60
    } else {
        $response = $callback
    }
    [ordered]@{
        name = $case.name
        workflow = $case.workflow
        input = $case.input
        expected = $case.expected
        status = [int]$response.StatusCode
        body = $response.Content
        passed = $response.StatusCode -eq 200 -and $response.Content -ceq $case.expected
    }
}
$definitions = foreach ($name in ($cases.workflow | Select-Object -Unique)) {
    $path = Join-Path (Join-Path $DefinitionsDirectory $name) 'workflow.json'
    [ordered]@{ name = $name; sha256 = (Get-FileHash -LiteralPath $path).Hash }
}
$assemblies = foreach ($relativePath in @(
    'bin\Microsoft.Azure.Workflows.Templates.dll',
    'bin\Microsoft.Azure.Workflows.Templates.Languages.Edge.CSharp.dll',
    'lib\codeful\Microsoft.Azure.Workflows.Sdk.dll',
    'lib\codeful\SourceNativeHostConsumer.dll'
)) {
    $path = Join-Path $HostDirectory $relativePath
    [ordered]@{
        path = $relativePath
        sha256 = (Get-FileHash -LiteralPath $path).Hash
        version = [Diagnostics.FileVersionInfo]::GetVersionInfo($path).FileVersion
    }
}
[ordered]@{
    version = 1
    evidenceKind = 'actual-local-logicapps-codeful-runtime'
    completedUtc = [DateTimeOffset]::UtcNow.ToString('o')
    runtime = $status.version
    processIds = @($listener.OwningProcess | Select-Object -Unique)
    definitions = @($definitions)
    assemblies = @($assemblies)
    results = @($results)
    limits = @('No cloud or designer certification.', 'Failed cases remain unsupported; a successful arithmetic probe does not approve every native capability.')
} | ConvertTo-Json -Depth 10 | Set-Content -LiteralPath $OutputPath -Encoding utf8
$results | ForEach-Object { [pscustomobject]$_ } | Format-Table name,status,passed
if (@($results | Where-Object { -not $_.passed }).Count) {
    throw "Native host contracts failed; exact observations are preserved in $OutputPath."
}
