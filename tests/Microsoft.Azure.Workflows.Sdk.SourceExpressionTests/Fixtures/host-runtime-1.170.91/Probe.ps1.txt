param(
    [Parameter(Mandatory)][string] $BaseUri,
    [Parameter(Mandatory)][string] $CultureEvidence,
    [Parameter(Mandatory)][string] $HostDirectory,
    [Parameter(Mandatory)][string] $OutputPath
)
$ErrorActionPreference = 'Stop'
$uri = [Uri]$BaseUri
if (-not $uri.IsLoopback) { throw 'This probe is restricted to an isolated local host.' }
$culture = Get-Content -LiteralPath $CultureEvidence -Raw | ConvertFrom-Json
$listeners = @(Get-NetTCPConnection -State Listen -LocalPort $uri.Port)
if ($culture.processId -notin $listeners.OwningProcess) {
    throw 'Culture evidence does not identify the process serving the host endpoint.'
}
$status = Invoke-RestMethod "$BaseUri/admin/host/status"
if ($status.state -ne 'Running') { throw "Host is not running: $($status.state)" }
$callbackPath = '/runtime/webhooks/workflow/api/management/workflows/{0}/triggers/manual/listCallbackUrl?api-version=2018-11-01'
$template = Invoke-RestMethod -Method Post ($BaseUri + ($callbackPath -f 'TemplateProbe'))
$cases = @(
    @{ name = 'null'; input = 'null'; expected = 'Received request: ' },
    @{ name = 'string'; input = '"hello"'; expected = 'Received request: hello' },
    @{ name = 'integer'; input = '42'; expected = 'Received request: 42' },
    @{ name = 'boolean'; input = 'true'; expected = 'Received request: True' },
    @{ name = 'object'; input = '{"n":1}'; expected = 'Received request: {"n":1}' },
    @{ name = 'array'; input = '[1,2]'; expected = 'Received request: [1,2]' }
)
$results = foreach ($case in $cases) {
    $response = Invoke-WebRequest -Method Post $template.value -ContentType 'application/json' -Body $case.input
    [ordered]@{ inputKind = $case.name; inputJson = $case.input; status = [int]$response.StatusCode; body = $response.Content; passed = $response.StatusCode -eq 200 -and $response.Content -ceq $case.expected }
}
$literal = Invoke-RestMethod -Method Post ($BaseUri + ($callbackPath -f 'LiteralProbe'))
$response = Invoke-WebRequest -Method Post $literal.value -ContentType 'application/json' -Body '{}' -SkipHttpErrorCheck
$literalPassed = $response.StatusCode -eq 200 -and $response.Content -ceq '@csharp{1 + 2}'
$definitions = foreach ($name in @('TemplateProbe','LiteralProbe','NativeProbe','ConditionProbe')) {
    $path = Join-Path (Join-Path $HostDirectory $name) 'workflow.json'
    [ordered]@{ name = $name; sha256 = (Get-FileHash -LiteralPath $path).Hash }
}
$evidence = [ordered]@{
    version = 1
    completedUtc = [DateTimeOffset]::UtcNow.ToString('o')
    evidenceKind = 'actual-local-logicapps-runtime'
    runtime = $status.version
    extensionBundle = $status.extensionBundle
    cultureInstrumentation = $culture
    definitions = @($definitions)
    templateResults = @($results)
    literalResult = [ordered]@{ passed = $literalPassed; status = [int]$response.StatusCode; body = $response.Content }
    limits = @('No cloud or designer certification.', 'Native-expression support must be assessed separately; successful literal transport does not prove it.')
}
$evidence | ConvertTo-Json -Depth 10 | Set-Content -LiteralPath $OutputPath -Encoding utf8
$results | ForEach-Object { [pscustomobject]$_ } | Format-Table inputKind,status,body
if (-not $literalPassed) { throw "SDK literal marker transport failed; exact response and definition hashes are preserved in $OutputPath." }
if (@($results | Where-Object { -not $_.passed }).Count) { throw "Template formatting differs from the measured host contract; exact results are preserved in $OutputPath." }
