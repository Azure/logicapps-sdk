using Microsoft.Azure.Workflows.Sdk;

internal static class FixtureWorkflow
{
    public static object Run()
    {
        var count = WorkflowActions.BuiltIn.Compose<int>(() => 1).WithName("Count");
        var action = WorkflowActions.BuiltIn.Compose<int>(() => count.Output + 1);
        bool overflow = false;
        int maximum = int.MaxValue;
        try { _ = maximum + 1; }
        catch (OverflowException) { overflow = true; }
        return new { input = FixtureAssert.Input(action), overflow };
    }
}
